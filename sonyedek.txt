def calculate_position_size(risk_amount, stop_distance):
    """Pozisyon büyüklüğü hesaplama"""
    try:
        if stop_distance <= 0:
            print("❌ Stop mesafesi 0 veya negatif olamaz!")
            return 0
        
        position_size = risk_amount / stop_distance
        
        # Minimum ve maksimum pozisyon büyüklüğü kontrolleri
        if position_size < 0.001:  # Minimum 0.001 BTC
            position_size = 0.001
            print("ℹ️ Pozisyon büyüklüğü minimum değere ayarlandı: 0.001")
        elif position_size > 0.1:  # Maksimum 0.1 BTC
            position_size = 0.1
            print("ℹ️ Pozisyon büyüklüğü maksimum değere ayarlandı: 0.1")
            
        return round(position_size, 3)
    except Exception as e:
        print(f"❌ Pozisyon büyüklüğü hesaplama hatası: {e}")
        return 0.001  # Hata durumunda minimum değeri kullan

def calculate_tp1(entry, size):
    """İlk take profit seviyesi"""

def detect_sweep_reversal(data, level):
    """Sweep ve Reversal pattern tespiti"""
    try:
        # Level değerini çıkar
        if isinstance(level, dict):
            if "price" in level:
                level_price = level["price"]
            elif "high" in level and "low" in level:
                level_price = (level["high"] + level["low"]) / 2
            else:
                return False
        elif isinstance(level, tuple):
            level_price = level[0]  # Premium zone'u kullan
        elif isinstance(level, (int, float)):
            level_price = float(level)
        else:
            return False
            
        for i in range(4, len(data)-1):
            # Aşağı sweep
            if (data['low'].iloc[i] < level_price and
                data['close'].iloc[i] > data['open'].iloc[i] and
                data['volume'].iloc[i] > data['volume'].iloc[i-1]):
                return True
            # Yukarı sweep
            if (data['high'].iloc[i] > level_price and
                data['close'].iloc[i] < data['open'].iloc[i] and
                data['volume'].iloc[i] > data['volume'].iloc[i-1]):
                return True
        return False
    except Exception as e:
        print(f"Sweep reversal tespiti hatası: {e}")
        return False

def calculate_fair_value_nodes(data):
    """FVN (Fair Value Node) hesaplama"""
    median = data['close'].median()
    std = data['close'].std()
    premium_zone = median + (std * 1.5)
    discount_zone = median - (std * 1.5)
    return premium_zone, discount_zone

# Market Structure
def identify_market_structure(data):
    """Market yapısını analiz et (Trend/Range)"""
    try:
        # Son 20 mum için yüksek ve alçak noktaları bul
        highs = data['high'].rolling(window=5).max()
        lows = data['low'].rolling(window=5).min()
        
        # Higher Highs (HH) ve Lower Lows (LL) sayısını hesapla
        hh_count = sum(highs.diff() > 0)
        ll_count = sum(lows.diff() < 0)
        
        # Trend tespiti
        if hh_count > ll_count * 1.5:
            return "BULLISH_TREND"
        elif ll_count > hh_count * 1.5:
            return "BEARISH_TREND"
        else:
            return "RANGE"
            
    except Exception as e:
        print(f"❌ Market yapısı analizi hatası: {e}")
        return "UNKNOWN"

def calculate_fvg(data):
    """Fair Value Gap hesaplama"""
    fvg_list = []
    
    for i in range(1, len(data)-1):
        # Bullish FVG
        if data['low'].iloc[i+1] > data['high'].iloc[i-1]:
            fvg_list.append({
                "type": "BULLISH",
                "top": data['low'].iloc[i+1],
                "bottom": data['high'].iloc[i-1],
                "index": i,
                "timestamp": data['timestamp'].iloc[i]  # Timestamp eklendi
            })
        # Bearish FVG
        elif data['high'].iloc[i+1] < data['low'].iloc[i-1]:
            fvg_list.append({
                "type": "BEARISH",
                "top": data['low'].iloc[i-1],
                "bottom": data['high'].iloc[i+1],
                "index": i,
                "timestamp": data['timestamp'].iloc[i]  # Timestamp eklendi
            })
    
    return fvg_list

def detect_sibi(data):
    """Supply Imbalance, Breaker, Inefficiency tespiti"""
    sibi_zones = []
    
    for i in range(2, len(data)-2):
        # Supply Imbalance
        if data['low'].iloc[i] > data['high'].iloc[i-1]:
            sibi_zones.append({
                "type": "SUPPLY_IMBALANCE",
                "top": data['low'].iloc[i],
                "bottom": data['high'].iloc[i-1],
                "index": i
            })
        
        # Demand Imbalance
        if data['high'].iloc[i] < data['low'].iloc[i-1]:
            sibi_zones.append({
                "type": "DEMAND_IMBALANCE",
                "top": data['low'].iloc[i-1],
                "bottom": data['high'].iloc[i],
                "index": i
            })
    
    return sibi_zones

def calculate_pdh_pdl(data):
    """Previous Day High/Low hesaplama"""
    data['datetime'] = pd.to_datetime(data['timestamp'], unit='ms')
    data['date'] = data['datetime'].dt.date
    
    daily_high = data.groupby('date')['high'].max()
    daily_low = data.groupby('date')['low'].min()
    
    return {
        "pdh": daily_high.iloc[-2],
        "pdl": daily_low.iloc[-2]
    }

            # 1. Üç ardışık aynı yönlü mum
            bullish_sequence = all(data['close'].iloc[j] > data['open'].iloc[j] for j in range(i-2, i+1))
            bearish_sequence = all(data['close'].iloc[j] < data['open'].iloc[j] for j in range(i-2, i+1))
            
            if bullish_sequence or bearish_sequence:
                # 2. Hacim artışı
                volume_increasing = data['volume'].iloc[i] > data['volume'].iloc[i-1] * 1.5
                
                # 3. Momentum kontrolü
                momentum = (data['close'].iloc[i] - data['open'].iloc[i-2]) / data['open'].iloc[i-2]
                strong_momentum = abs(momentum) > 0.003  # %0.3
                
                if volume_increasing and strong_momentum:
                    return {
                        "type": "BULLISH" if bullish_sequence else "BEARISH",
                        "level": data['close'].iloc[i],
                        "volume_confirmation": True,
                        "momentum": momentum
                    }
        return None
    except Exception as e:
        print(f"❌ Silver Bullet tespit hatası: {e}")
        return None

def detect_unicorn_model(data):
    """Unicorn Model pattern tespiti"""
    try:
        for i in range(4, len(data)-1):
            # 1. Market yapısı kırılımı
            structure_break = False
            direction = None
            
            # Bullish yapı kırılımı
            if (data['low'].iloc[i] > data['high'].iloc[i-1] and 
                data['low'].iloc[i-1] > data['high'].iloc[i-2]):
                structure_break = True
                direction = "BULLISH"
                
            # Bearish yapı kırılımı
            elif (data['high'].iloc[i] < data['low'].iloc[i-1] and 
                  data['high'].iloc[i-1] < data['low'].iloc[i-2]):
                structure_break = True
                direction = "BEARISH"
            
            if structure_break:
                # 2. Retest kontrolü
                retest_level = data['high'].iloc[i-1] if direction == "BEARISH" else data['low'].iloc[i-1]
                
                # 3. Hacim kontrolü
                volume_confirmation = data['volume'].iloc[i] > data['volume'].rolling(20).mean().iloc[i]

                    return {
                        "direction": direction,
                        "level": retest_level,
                        "structure_break": True,
                        "retest_level": retest_level
                    }
        return None
    except Exception as e:
        print(f"❌ Unicorn Model tespit hatası: {e}")
        return None

def detect_market_maker_patterns(data):
    """Market Maker Patterns (AMD, Buy/Sell Models)"""
    mm_patterns = []
    
    for i in range(4, len(data)-1):
        # Accumulation Model
        if (data['volume'].iloc[i] > data['volume'].iloc[i-4:i].mean() * 2 and
            abs(data['close'].iloc[i] - data['open'].iloc[i]) < abs(data['high'].iloc[i] - data['low'].iloc[i]) * 0.3):
            mm_patterns.append({
                "type": "ACCUMULATION",
                "price": data['close'].iloc[i],
                "volume": data['volume'].iloc[i],
                "index": i
            })
        # Distribution Model
        elif (data['volume'].iloc[i] > data['volume'].iloc[i-4:i].mean() * 2 and
              data['close'].iloc[i] < data['open'].iloc[i]):
            mm_patterns.append({
                "type": "DISTRIBUTION",

                "volume": data['volume'].iloc[i],
                "index": i
            })
    
    return mm_patterns

def analyze_iof_filters(data):
    """Institutional Order Flow Filtreleri"""
    iof_signals = []
    
    for i in range(3, len(data)-1):
        avg_volume = data['volume'].iloc[i-3:i].mean()
        delta = (data['close'].iloc[i] - data['open'].iloc[i]) * data['volume'].iloc[i]
        
        if data['volume'].iloc[i] > avg_volume * 2:  # Yüksek hacim
            if delta > 0:  # Alım baskısı
                iof_signals.append({
                    "type": "INSTITUTIONAL_BUY",
                    "price": data['close'].iloc[i],
                    "volume": data['volume'].iloc[i],
                    "delta": delta,
                    "index": i
                })
            else:  # Satış baskısı
                iof_signals.append({
                    "type": "INSTITUTIONAL_SELL",
                    "price": data['close'].iloc[i],
                    "volume": data['volume'].iloc[i],
                    "delta": delta,
                    "index": i
                })
    
    return iof_signals

def detect_vector_candle(data, level):
    """Vector candle pattern tespiti"""
    try:
        # Level değerini çıkar
        if isinstance(level, dict):
            if "price" in level:
                level_price = level["price"]
            elif "high" in level and "low" in level:
                level_price = (level["high"] + level["low"]) / 2
            else:
                return False
        elif isinstance(level, tuple):
            level_price = level[0]
        elif isinstance(level, (int, float)):
            level_price = float(level)
        else:
            return False
            
        for i in range(3, len(data)-1):
            # Bullish vector candle
            if (data['close'].iloc[i] > data['open'].iloc[i] and
                data['close'].iloc[i] > level_price and
                data['volume'].iloc[i] > data['volume'].iloc[i-3:i].mean() * 1.5 and
                (data['close'].iloc[i] - data['open'].iloc[i]) / (data['high'].iloc[i] - data['low'].iloc[i]) > 0.6):
                return True
            
            # Bearish vector candle
            if (data['close'].iloc[i] < data['open'].iloc[i] and
                data['close'].iloc[i] < level_price and
                data['volume'].iloc[i] > data['volume'].iloc[i-3:i].mean() * 1.5 and
                (data['open'].iloc[i] - data['close'].iloc[i]) / (data['high'].iloc[i] - data['low'].iloc[i]) > 0.6):
                return True
        return False
    except Exception as e:
        print(f"Vector candle tespiti hatası: {e}")
        return False

def detect_sniper_entry(data):
    """ICT Sniper Entry modellerini tespit et"""
    sniper_setups = []
    
    for i in range(4, len(data)-1):
        # Model 1: Sweep ve Reversal
        if (data['low'].iloc[i] < min(data['low'].iloc[i-4:i]) and  # Sweep down
            data['close'].iloc[i] > data['open'].iloc[i] and        # Bullish close
            data['volume'].iloc[i] > data['volume'].iloc[i-1]):     # Volume confirmation
            sniper_setups.append({
                "type": "BULLISH_SWEEP_REVERSAL",
                "entry": data['close'].iloc[i],
                "stop": data['low'].iloc[i],
                    "index": i
                })
        
        elif (data['high'].iloc[i] > max(data['high'].iloc[i-4:i]) and  # Sweep up
              data['close'].iloc[i] < data['open'].iloc[i] and          # Bearish close
              data['volume'].iloc[i] > data['volume'].iloc[i-1]):       # Volume confirmation
            sniper_setups.append({
                "type": "BEARISH_SWEEP_REVERSAL",
                "entry": data['close'].iloc[i],
                "stop": data['high'].iloc[i],
                    "index": i
                })
        
        # Model 2: Liquidity Void Entry
        if (data['low'].iloc[i] > data['high'].iloc[i-1] and    # Gap up
            data['volume'].iloc[i] < data['volume'].iloc[i-1]):  # Lower volume
            sniper_setups.append({
                "type": "VOID_LONG",
                "entry": data['low'].iloc[i],
                "stop": data['low'].iloc[i-1],
                "index": i
            })
            
        elif (data['high'].iloc[i] < data['low'].iloc[i-1] and   # Gap down
              data['volume'].iloc[i] < data['volume'].iloc[i-1]): # Lower volume
            sniper_setups.append({
                "type": "VOID_SHORT",
                "entry": data['high'].iloc[i],
                "stop": data['high'].iloc[i-1],
                "index": i
            })
        
        # Model 3: Engineered Move Entry
        if (all(data['close'].iloc[i-j] > data['open'].iloc[i-j] for j in range(3)) and  # 3 bullish candles
            data['volume'].iloc[i] > data['volume'].iloc[i-3:i].mean() * 1.5):           # Volume spike
            sniper_setups.append({
                "type": "ENGINEERED_REVERSAL_SHORT",
                "entry": data['close'].iloc[i],
                "stop": data['high'].iloc[i] + (data['high'].iloc[i] - data['low'].iloc[i]) * 0.1,
                "index": i
            })
            
        elif (all(data['close'].iloc[i-j] < data['open'].iloc[i-j] for j in range(3)) and  # 3 bearish candles
              data['volume'].iloc[i] > data['volume'].iloc[i-3:i].mean() * 1.5):           # Volume spike
            sniper_setups.append({
                "type": "ENGINEERED_REVERSAL_LONG",
                "entry": data['close'].iloc[i],
                "stop": data['low'].iloc[i] - (data['high'].iloc[i] - data['low'].iloc[i]) * 0.1,
                "index": i
            })
    
    return sniper_setups

def calculate_tp_by_rr(entry, sl, direction):
    """Risk/Ödül oranına göre Take Profit hesapla"""
    risk = abs(entry - sl)
    if direction == "LONG":
        return entry + (risk * risk_reward_ratio)
    else:  # SHORT
        return entry - (risk * risk_reward_ratio)

def fresh_order_block(obs):
    """Yeni oluşmuş Order Block kontrolü"""
    if not obs:
        return False
    latest_ob = obs[-1]
    current_time = time.time() * 1000
    return (current_time - latest_ob["timestamp"]) < 3600000  # Son 1 saat içinde

def active_breaker(breakers):
    """Aktif Breaker Block kontrolü"""
    if not breakers:
        return False
    latest_breaker = breakers[-1]
    # broken özelliği yerine price kontrolü yapalım
    current_price = float(data["ltf"]['close'].iloc[-1])
    if "price" in latest_breaker:
        return current_price < latest_breaker["price"]  # Fiyat breaker seviyesinin altındaysa aktif

def rank_opportunities(entries):
    """Entry fırsatlarını önceliklendir"""
    try:
        if not entries:
            return []
            
        ranked_entries = []
        for entry in entries:
            score = 0
            
            # 1. Setup tipine göre puan
            if entry["setup_type"] == "PREMIUM_DISCOUNT":
                score += 3
            elif entry["setup_type"] == "ORDER_BLOCK":
                score += 2
            elif entry["setup_type"] == "FVG":
                score += 1
                
            # 2. Entry tipine göre puan
            if entry["type"] == "SWEEP_REVERSAL":
                score += 3
            elif entry["type"] == "LIQUIDITY_TEST":
                score += 2
            elif entry["type"] == "MICRO_FAILURE":
                score += 1
                
            # 3. Risk/Ödül oranına göre puan
            rr_ratio = abs(entry["entry"] - calculate_tp1(entry, 0)) / abs(entry["entry"] - entry["stop"])
            if rr_ratio >= 3:
                score += 3
            elif rr_ratio >= 2:
                score += 2
            elif rr_ratio >= 1.5:
                score += 1
                
            ranked_entries.append((score, entry))
            
        # Puana göre sırala ve en iyi 3 fırsatı döndür
        ranked_entries.sort(reverse=True)
        return [entry for score, entry in ranked_entries[:3]]
        
    except Exception as e:
        print(f"❌ Fırsat sıralama hatası: {e}")
        return entries  # Hata durumunda orijinal listeyi döndür

def validate_entry_conditions(entry):
    """Entry koşullarını doğrula"""
    try:
        current_price = float(data["scalp"]['close'].iloc[-1])
        
        # 1. Fiyat entry seviyesine yakın mı?
        if abs(current_price - entry["entry"]) / current_price > 0.001:
            return False
        
        # 2. Stop loss mesafesi uygun mu?
        stop_distance = abs(entry["entry"] - entry["stop"])
        if stop_distance / current_price > 0.005:  # Max %0.5 risk
            return False
        
        # 3. Hacim onayı var mı?
        if data["scalp"]['volume'].iloc[-1] < data["scalp"]['volume'].iloc[-2]:
            return False
        
        # 4. Yön onayı var mı?
        if entry["type"].startswith("BULLISH") and data["scalp"]['close'].iloc[-1] < data["scalp"]['open'].iloc[-1]:
            return False
        if entry["type"].startswith("BEARISH") and data["scalp"]['close'].iloc[-1] > data["scalp"]['open'].iloc[-1]:
            return False
        
        return True
    except Exception as e:
        print(f"Entry koşulları kontrolü hatası: {e}")
        return False

def calculate_momentum(data):
    """Momentum göstergelerini hesapla"""
    try:
        # RSI hesaplama (14 periyot)
        delta = data['close'].diff()
        gain = (delta.where(delta > 0, 0)).rolling(window=14).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(window=14).mean()
        rs = gain / loss
        rsi = 100 - (100 / (1 + rs))
        
        # MACD hesaplama
        exp1 = data['close'].ewm(span=12, adjust=False).mean()
        exp2 = data['close'].ewm(span=26, adjust=False).mean()
        macd = exp1 - exp2
        signal = macd.ewm(span=9, adjust=False).mean()
        
        return {
            "rsi": rsi.iloc[-1],
            "macd": macd.iloc[-1],
            "macd_signal": signal.iloc[-1],
            "trend": "BULLISH" if macd.iloc[-1] > signal.iloc[-1] else "BEARISH"
        }
        
    except Exception as e:
        print(f"❌ Momentum hesaplama hatası: {e}")
        return None

def calculate_volatility(data):
    """Volatilite hesaplama"""
    try:
        # ATR (Average True Range) hesaplama
        high_low = data['high'] - data['low']
        high_close = abs(data['high'] - data['close'].shift())
        low_close = abs(data['low'] - data['close'].shift())
        ranges = pd.concat([high_low, high_close, low_close], axis=1)
        true_range = ranges.max(axis=1)
        atr = true_range.rolling(14).mean()
        
        # Bollinger Bands
        sma = data['close'].rolling(window=20).mean()
        std = data['close'].rolling(window=20).std()
        upper_band = sma + (std * 2)
        lower_band = sma - (std * 2)
        
        current_price = data['close'].iloc[-1]
        bb_position = (current_price - lower_band.iloc[-1]) / (upper_band.iloc[-1] - lower_band.iloc[-1])
        
        return {
            "atr": atr.iloc[-1],
            "bb_position": bb_position,
            "volatility_state": "HIGH" if atr.iloc[-1] > atr.mean() else "LOW"
        }
        
    except Exception as e:
        print(f"❌ Volatilite hesaplama hatası: {e}")
        return None

def detect_candlestick_patterns(data):
    """Mum formasyonlarını tespit et"""
    try:
        patterns = []
        
        for i in range(1, len(data)-1):
            body = abs(data['close'].iloc[i] - data['open'].iloc[i])
            upper_wick = data['high'].iloc[i] - max(data['open'].iloc[i], data['close'].iloc[i])
            lower_wick = min(data['open'].iloc[i], data['close'].iloc[i]) - data['low'].iloc[i]
            
            # Doji
            if body < (data['high'].iloc[i] - data['low'].iloc[i]) * 0.1:
                patterns.append({
                    "type": "DOJI",
                    "index": i,
                    "strength": "MEDIUM"
                })
                
            # Pin Bar
            elif (upper_wick > body * 2 or lower_wick > body * 2) and body > 0:
                patterns.append({
                    "type": "PIN_BAR",
                    "index": i,
                    "strength": "HIGH"
                })
                
            # Engulfing
            elif i > 0:
                prev_body = abs(data['close'].iloc[i-1] - data['open'].iloc[i-1])
                if body > prev_body:
                    if (data['close'].iloc[i] > data['open'].iloc[i] and 
                        data['close'].iloc[i-1] < data['open'].iloc[i-1]):
                        patterns.append({
                            "type": "BULLISH_ENGULFING",
                            "index": i,
                            "strength": "HIGH"
                        })
                    elif (data['close'].iloc[i] < data['open'].iloc[i] and 
                          data['close'].iloc[i-1] > data['open'].iloc[i-1]):
                        patterns.append({
                            "type": "BEARISH_ENGULFING",
                            "index": i,
                            "strength": "HIGH"
                        })
        
        return patterns
        
    except Exception as e:
        print(f"❌ Mum formasyonu tespiti hatası: {e}")
        return []

def calculate_level_strength(data, index):
    """Fiyat seviyesinin gücünü hesapla"""
    try:
        # Hacim kontrolü
        volume_strength = 1.0
        if data['volume'].iloc[index] > data['volume'].rolling(20).mean().iloc[index]:
            volume_strength = 1.5
            
        # Önceki testler
        price_level = data['high'].iloc[index] if data['high'].iloc[index] > data['high'].iloc[index-1] else data['low'].iloc[index]
        test_count = 0
        bounce_count = 0
        
        for i in range(max(0, index-20), min(len(data), index+20)):
            if abs(data['high'].iloc[i] - price_level) / price_level < 0.001 or abs(data['low'].iloc[i] - price_level) / price_level < 0.001:
                test_count += 1
                if i < index and ((data['close'].iloc[i] > data['open'].iloc[i] and data['low'].iloc[i] > price_level) or 
                                (data['close'].iloc[i] < data['open'].iloc[i] and data['high'].iloc[i] < price_level)):
                    bounce_count += 1
                    
        level_strength = (1 + (test_count * 0.1) + (bounce_count * 0.2)) * volume_strength
        
        return round(level_strength, 2)
        
    except Exception as e:
        print(f"❌ Seviye gücü hesaplama hatası: {e}")
        return 1.0

def detect_fvg():
    """Fair Value Gap tespit et"""
    try:
        fvg = None
        htf_data = data["htf"].copy()
        
        # Timestamp'i index olarak ayarla ve indexi resetle
        htf_data = htf_data.reset_index(drop=True)
        
        # Volatilite hesapla
        volatility = calculate_volatility(htf_data)
        
        # HTF'de FVG ara
        for i in range(1, len(htf_data)-1):
            # Bullish FVG
            if htf_data['low'].iloc[i+1] > htf_data['high'].iloc[i-1]:
                gap_size = float(htf_data['low'].iloc[i+1] - htf_data['high'].iloc[i-1])
                fvg = {
                    "type": "BULLISH_FVG",
                    "top": float(htf_data['low'].iloc[i+1]),
                    "bottom": float(htf_data['high'].iloc[i-1]),
                    "size": gap_size,
                    "atr": volatility["atr"]
                }
                break
            # Bearish FVG
            elif htf_data['high'].iloc[i+1] < htf_data['low'].iloc[i-1]:
                gap_size = float(htf_data['low'].iloc[i-1] - htf_data['high'].iloc[i+1])
                fvg = {
                    "type": "BEARISH_FVG",
                    "top": float(htf_data['low'].iloc[i-1]),
                    "bottom": float(htf_data['high'].iloc[i+1]),
                    "size": gap_size,
                    "atr": volatility["atr"]
                }
                break
        return fvg
    except Exception as e:
        print(f"❌ FVG tespit hatası: {e}")
        return None

def calculate_volatility(df):
    """Volatilite hesaplama"""
    try:
        # ATR (Average True Range) hesaplama
        df = df.copy()
        df['high_low'] = df['high'] - df['low']
        df['high_close'] = abs(df['high'] - df['close'].shift())
        df['low_close'] = abs(df['low'] - df['close'].shift())
        
        true_range = df[['high_low', 'high_close', 'low_close']].max(axis=1)
        atr = true_range.rolling(window=14, min_periods=1).mean()
        
        current_volatility = "HIGH" if atr.iloc[-1] > atr.mean() else "LOW"
        
        return {
            "atr": float(atr.iloc[-1]),
            "volatility_state": current_volatility
        }
        
    except Exception as e:
        print(f"❌ Volatilite hesaplama hatası: {e}")
        return {"atr": 0.0, "volatility_state": "UNKNOWN"}

def detect_pd_zones():
    """Premium/Discount bölgelerini tespit et"""
    try:
        htf_data = data["htf"].copy()
        
        # VWAP hesapla
        htf_data['vwap'] = (htf_data['close'] * htf_data['volume']).cumsum() / htf_data['volume'].cumsum()
        std = htf_data['close'].std()
        
        premium_zone = float(htf_data['vwap'].iloc[-1] + (std * 2))
        discount_zone = float(htf_data['vwap'].iloc[-1] - (std * 2))
        
        zones = []
        zones.append({
            "type": "PREMIUM",
            "level": premium_zone,
            "strength": "HIGH"
        })
        
        zones.append({
            "type": "DISCOUNT",
            "level": discount_zone,
            "strength": "HIGH"
        })
        
        return zones
    except Exception as e:
        print(f"❌ PD Zones tespit hatası: {e}")
        return []
